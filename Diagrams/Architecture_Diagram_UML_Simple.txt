@startuml
title Decentralized Compute Marketplace - System Architecture

left to right direction

actor "Compute Consumer" as Consumer
actor "Compute Provider" as Provider

package "Consumer Environment" {
    component "Consumer App" as ConsumerApp
}

package "Blockchain Network" {

    component "Marketplace Factory\nSmart Contract" as Factory

    component "Job / Escrow\nSmart Contract" as Job

    component "Provider Registry\nSmart Contract" as Registry

    note right of Job
      Responsibilities:
      • Job state
      • Provider participation
      • Verification
      • Escrow
      • Payment
      • Penalties
    end note
}

package "Provider Environment" {

    component "Provider App" as ProviderApp

    component "Container Runtime" as Runtime
}

database "Distributed Storage" as Storage

Consumer --> ConsumerApp

Provider --> ProviderApp

ConsumerApp --> Factory : Create job + escrow
Factory --> Job : Deploy job contract

ConsumerApp --> Job : Monitor job / retrieve status

ProviderApp --> Registry : Register / stake
ProviderApp --> Job : Claim / commit / reveal /\nverify / receive payment

Job --> Registry : Check eligibility / stake

ProviderApp --> Runtime : Execute workload

ConsumerApp --> Storage : Upload input /\nretrieve result
ProviderApp --> Storage : Download input /\nupload result

@enduml
