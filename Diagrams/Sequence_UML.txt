autonumber

actor Buyer
participant "React Web App" as Web
box "Server" #FFF6E5
    participant "Server\n(API + Matcher +\nBlockchain Mgr)" as Server
    database "MongoDB" as DB
end box
box "Blockchain" #EAF7EA
    participant "JobEscrow" as Escrow
    participant "ProviderRegistry" as Registry
end box
participant "Provider Daemon\n(x k)" as Daemon

== Provider onboarding (once per provider) ==

Daemon -> Daemon : createSessionKey()
Daemon -> Registry : registerProvider(rate, sessionKey) + stake
Daemon -> Server : POST /provider/machineSpecs (benchmark scores)
Server -> DB : save Hardware
Daemon -> Server : TLS connect: HELLO(wallet, sessionKey)
Server -> Server : add to provider pool cache
Daemon -> Server : HEARTBEAT (periodic)

== Job submission ==

Buyer -> Web : fill form (Docker image, maxPrice,\nk, quorum, window, min specs)
Web -> Server : POST /jobs (JWT)
Server -> DB : save Job (PENDING) + DockerImage
Server -> Escrow : submitJob(buyer, imageHash, maxPrice,\nk, quorum, window) + escrow
Escrow --> Server : event JobSubmitted(jobId)
Server -> Server : findTopK(job, k)

== Assignment (off-chain match, on-chain accept) ==

loop for each of the top-k providers
    Server -> Daemon : TLS ASSIGN_JOB(jobId)
    Daemon -> Registry : addStake() if needed
    Daemon -> Escrow : acceptJob(jobId, sessionSignature)
    Escrow -> Registry : isEligible() + lockStake()
    Escrow --> Server : event JobAccepted(jobId, provider)\n[status PICKED_UP]
    Server -> Server : verify provider session
    Server -> Daemon : TLS DOCKER_IMAGE(jobId, image)
end
Escrow -> Escrow : k accepted -> RUNNING,\nstart commit window
Escrow --> Server : event JobStatusChanged(RUNNING)
Server -> DB : update Job.status

== Execution + commit ==

Daemon -> Daemon : runContainer(image, fixed seed)\nresultHash = hash(result)
Daemon -> Escrow : submitResultHash(jobId,\nkeccak(resultHash, nonce), sig)
Daemon -> Server : TLS RESULT(jobId, full result)
Server -> DB : store ProviderResult

== Reveal ==

Escrow -> Escrow : commit window closes -> VERIFYING
Daemon -> Escrow : submitResultDecoding(jobId,\nresultHash, nonce, sig)
Escrow -> Escrow : check keccak(resultHash, nonce) == commitHash

== Consensus + settlement ==

note over Escrow
  Triggered by the last reveal, or by
  finalizeTimeout() after the reveal deadline
end note
Escrow -> Escrow : _resolveConsensus(jobId)

alt largest group of matching hashes >= quorum
    Escrow -> Escrow : _payWinnersAndSlashLosers()
    Escrow --> Daemon : pay winners (maxPrice split by rate)
    Escrow -> Registry : unlockStake(winners)\nslash(losers / non-revealers)
    Escrow --> Server : event JobCompleted(jobId, consensusHash)
    Server -> DB : Job.status = COMPLETED
    Server -> DB : pick ProviderResult with\nresultHash == consensusHash
    Buyer -> Web : view job
    Web -> Server : GET /jobs/:id
    Server --> Web : consensus result
    Web --> Buyer : result
else no consensus (non-deterministic / failed job)
    Escrow -> Registry : slash(all providers)
    Escrow -> Escrow : buyer escrow forfeited
    Escrow --> Server : event JobFailed(jobId)
    Server -> DB : Job.status = FAILED
    Web --> Buyer : job failed
end
@enduml
