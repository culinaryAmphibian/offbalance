package "Provider VM  (VMware image, no GUI)" #F3ECFA {

    class VMConfig <<config>> {
        + cpuCores: int
        + memoryMB: long
        + gpuEnabled: boolean
        + storageMB: long
        + walletAddress: String
        + serverHost: String
        + backdoorPort: int
    }

    class ProviderDaemon {
        - config: VMConfig
        - protocol: ProtocolManager
        - chain: ProviderBlockchainManager
        - benchmark: Benchmark
        - executor: JobExecutor
        + start()
        + stop()
        - onAssignment(jobId)
        - onDockerImage(jobId, image)
    }

    class ProtocolManager {
        - socket: TLSSocket
        - port: int
        + connect(host, port)
        + sendHello(wallet, sessionKey, specs)
        + sendHeartbeat()
        + handleMessage(msg: ProtocolMessage)
        + sendAcceptAck(jobId)
        + sendResult(jobId, result)
    }

    class ProtocolMessage {
        + type: MessageType
        + jobId: uint
        + payload: bytes
        + signature: bytes
    }

    enum MessageType {
        HELLO
        HEARTBEAT
        ASSIGN_JOB
        ACCEPT_ACK
        DOCKER_IMAGE
        RESULT
        ERROR
    }

    class ProviderBlockchainManager {
        - wallet: address
        {field} - sessionKey: PrivateKey  (env var)
        - jobEscrow: Contract
        - providerRegistry: Contract
        - nonces: Map<jobId, bytes32>
        + createSessionKey()
        + ensureRegistered(rate)
        + ensureStaked(requiredStake)
        + acceptJob(jobId)
        + commitResult(jobId, resultHash)
        + revealResult(jobId)
        + waitForRevealPhase(jobId)
    }

    class Benchmark {
        + run() : BenchmarkScores
        + postMachineSpecs(scores)
    }

    class BenchmarkScores {
        + cpuScore: float
        + gpuScore: float
        + memoryScore: float
    }

    class JobExecutor {
        - stateSeed: bytes32
        + loadImage(image) : ImageId
        + runContainer(imageId, limits) : Result
        + hashResult(result) : bytes32
        + cleanup(imageId)
    }
    note bottom of ProviderBlockchainManager
      Session key is created on first startup,
      registered in ProviderRegistry, and stays
      active until revoked. All on-chain calls
      from the daemon are signed with it.
    end note
    
    note bottom of JobExecutor
      Every container starts from the same
      fixed state seed so honest providers
      produce identical (hash-equal) results.
    end note
    
    note top of ProviderDaemon
      Job flow:
      ASSIGN_JOB -> ensureStaked() -> acceptJob()
      DOCKER_IMAGE -> runContainer() -> hashResult()
      -> commitResult() + sendResult() -> revealResult()
    end note
}

ProviderDaemon *-- VMConfig
ProviderDaemon *-- ProtocolManager
ProviderDaemon *-- ProviderBlockchainManager
ProviderDaemon *-- Benchmark
ProviderDaemon *-- JobExecutor
ProtocolManager ..> ProtocolMessage : sends / receives
ProtocolMessage --> MessageType
Benchmark ..> BenchmarkScores : produces

@enduml
